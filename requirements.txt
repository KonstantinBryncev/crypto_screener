# CCXT (CryptoCurrency eXchange Trading)
# Главный движок нашего бота. Это универсальная библиотека, которая стандартизирует 
# запросы к более чем 100 биржам. Вместо написания уникального кода под Bybit, 
# Binance и OKX, мы используем общий интерфейс ccxt. Мы будем применять её 
# асинхронную часть (ccxt.async_support), которая идеально экономит ресурсы ОЗУ и CPU.
ccxt>=4.4.0

# AIOHTTP (Asynchronous HTTP Client/Server)
# Базовая библиотека для выполнения асинхронных сетевых запросов в Python. 
# Нужна скринеру, чтобы опрашивать биржи параллельно и не блокировать поток 
# во время ожидания сетевых ответов. Также она пригодится нам для отправки 
# мгновенных алертов в Telegram-канал или Discord.
aiohttp>=3.10.0

# AIOHTTP-SOCKS (SOCKS Proxy Connector for aiohttp)
# Важнейшее расширение для реализации твоих планов по проксированию. Оно учит 
# библиотеки aiohttp и ccxt бесшовно работать через быстрые и безопасные 
# асинхронные SOCKS4/SOCKS5 прокси-серверы, которые ты поднимешь в будущем.
aiohttp-socks>=0.8.0

# PYTHON-DOTENV (Read key-value pairs from a .env file)
# Инструмент для безопасности данных. Позволяет вынести все конфиденциальные 
# настройки (API-ключи бирж, пароли, секретные токены Telegram-ботов) в отдельный 
# изолированный текстовый файл `.env` в корне проекта, не "зашивая" их прямо в код.
python-dotenv>=1.0.1
