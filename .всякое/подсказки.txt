# Лог-файлы
logoutput: syslog /etc/danted.log

# Интерфейсы и порты
# internal — адрес/интерфейс, который слушает прокси (0.0.0.0 — все адреса)
internal: 0.0.0.0 port = 1080

# external — сетевой интерфейс или IP, через который трафик пойдет в интернет.
# Замените eth0 на имя вашего внешнего интерфейса (проверить можно через ip a)
external: inet

# Методы аутентификации
# "username" означает проверку системных пользователей Linux (из /etc/passwd)
socksmethod: username
clientmethod: none

# Права для пользователей (под какими системными правами работает демон)
user.privileged: root
user.unprivileged: nobody

# Разрешаем клиентам подключаться к прокси-серверу
client pass {
	from: 0.0.0.0/0 to: 0.0.0.0/0
	log: connect disconnect error
}

# Блокируем трафик на loopback (localhost) для безопасности
socks block {
	from: 0.0.0.0/0 to: lo
	log: connect error
}

# Разрешаем авторизованным пользователям выходить в интернет через SOCKS
socks pass {
	from: 0.0.0.0/0 to: 0.0.0.0/0
	command: connect udpassociate
	log: connect disconnect error
	# Требовать обязательную авторизацию по логину/паролю
	socksmethod: username
}
